1"""Throttle async context manager + acquire/release lifecycle + keep-alive. 

2 

3Architecture sec. Public API Surface: ``async with layout.throttle(addr, long=True) as loco:``. 

4 

5Story 5.1 Task 0 spike (2026-05-21) found JMRI's throttle API is WebSocket-only and JMRI's 

6WS-level heartbeat keeps held throttles alive without per-throttle application heartbeats. 

7The keep-alive coroutine therefore ships as a no-op structural stub; Story 5.3 hardware

33class Throttle: 

34 """Async context manager for a JMRI throttle session. 

35 

36 A ``Throttle`` represents an acquired DCC decoder address on JMRI. The 

37 lifecycle is: 

38 

39 1. ``layout.throttle(addr, long=True)`` (or ``client.throttle(...)``) 

40 constructs an unacquired ``Throttle``. 

41 2. ``async with throttle as loco:`` issues the WS acquire and spawns 

42 a supervised keep-alive coroutine. 

43 3. On exit (clean or exceptional), the keep-alive is cancelled and a 

44 WS release envelope is sent. 

45 

46 **FR28 open-loop honesty.** A successful acquire means only that JMRI 

47 accepted the throttle session — it does NOT imply a locomotive at this 

48 DCC address is physically on the layout or responsive. NCE is open-loop 

49 with no DCC-bus feedback; pyjmri cannot detect "ghost throttle" (a held 

50 session for an absent locomotive). Control commands sent to a throttle 

51 for an absent address are silently accepted by JMRI and the booster. 

52 

53 Args: 

54 handle: The :class:`pyjmri._protocols.ClientHandle` (typically a 

55 :class:`pyjmri.Client`) that owns the underlying WS connection. 

56 dcc_address: DCC decoder address (1..10293 typical). 

57 long: ``True`` for long (4-digit) addressing, ``False`` for short. 

58 """

74 async def __aenter__(self) -> Self: 

75 """Acquire the throttle via WS and spawn the supervised keep-alive. 

76 

77 Successful acquire only means JMRI accepted the throttle; it does not 

78 imply a locomotive at this DCC address is physically on the layout or 

79 responsive — NCE is open-loop with no DCC-bus feedback (FR28). 

80 

81 Raises: 

82 ThrottleReleased: when this ``Throttle`` was already released. 

83 RuntimeError: when this ``Throttle`` is already acquired (i.e., 

84 ``__aenter__`` was called while a prior ``async with`` is 

85 still active on the same instance). 

86 ThrottleAcquireFailed: when JMRI rejects the acquire (e.g., 

87 invalid address). 

88 JMRIConnectionError, JMRIRequestTimeout: surfaced from the WS 

89 transport. 

90 """ 

91 if self._released: 

92 raise ThrottleReleased( 

93 "throttle already released; cannot re-acquire", 

94 dcc_address=self.dcc_address, 

95 ) 

96 if self._throttle_id is not None: 

97 raise RuntimeError( 

98 f"Throttle for DCC address {self.dcc_address} is already acquired; " 

99 "exit the current 'async with' block before re-entering" 

100 ) 

101 try: 

102 self._throttle_id = await self._handle.throttle_acquire( 

103 self.dcc_address, long=self.long 

104 ) 

105 except (JMRIConnectionError, JMRIRequestTimeout): 

106 raise 

107 except ThrottleAcquireFailed: 

108 raise 

109 self._keepalive_task = self._handle.spawn_supervised( 

110 self._keepalive(), 

111 name=f"pyjmri-throttle-keepalive-{self.dcc_address}", 

112 ) 

113 logger.info( 

114 "throttle acquired", 

115 extra={ 

116 "dcc_address": self.dcc_address, 

117 "throttle_id": self._throttle_id, 

118 "long": self.long, 

119 }, 

120 ) 

121 return self

123 async def __aexit__( 

124 self, 

125 exc_type: type[BaseException] | None, 

126 exc: BaseException | None, 

127 tb: TracebackType | None, 

128 ) -> None: 

129 body_raised = exc is not None 

130 if not self._released: 

131 await self._release_impl(suppress_errors=body_raised) 

132 # Return None: do not swallow the body exception.

134 async def release(self) -> None: 

135 """Cancel the keep-alive and send the WS release envelope. 

136 

137 After ``release()``, all control methods (``__aenter__``, 

138 :meth:`set_speed`, :meth:`set_function`) raise 

139 :class:`ThrottleReleased`. Release is idempotent — a second call 

140 returns without re-issuing the WS envelope. 

141 """ 

142 if self._released: 

143 return 

144 await self._release_impl(suppress_errors=False)

146 async def set_speed(self, value: float, *, forward: bool) -> None: 

147 """Set speed and direction in a single fire-and-forget WS update (FR25). 

148 

149 Sends ``{"type":"throttle","data":{"throttle":<id>,"speed":<value>, 

150 "forward":<forward>}}`` and returns as soon as the bytes are written; 

151 does NOT await JMRI's state-echo. JMRI emits per-field delta echoes 

152 which the WS dispatcher silently drops. 

153 

154 FR28 / NCE open-loop reminder: this method commands JMRI's view of 

155 the throttle's state. It does NOT confirm the physical locomotive 

156 responded — NCE has no DCC-bus feedback. A ghost throttle (an absent 

157 DCC address) silently accepts updates. 

158 

159 Args: 

160 value: Throttle setting in the closed range ``[0.0, 1.0]``. NaN 

161 and ``±inf`` are all rejected — the guard ``0.0 <= value 

162 <= 1.0`` evaluates ``False`` for NaN and for ``-inf``; 

163 ``inf > 1.0`` catches positive infinity. ``0.0`` is a 

164 valid emergency stop. 

165 forward: ``True`` for forward, ``False`` for reverse. 

166 

167 Raises: 

168 ThrottleReleased: when called after :meth:`release` (carries 

169 ``dcc_address`` in ``.context``). No WS traffic is sent. 

170 RuntimeError: when called on a never-acquired ``Throttle`` 

171 (the ``async with`` block has not been entered). 

172 ValueError: when ``value`` is outside ``[0.0, 1.0]`` or is NaN. 

173 JMRIConnectionError: surfaced from the WS transport. 

174 """ 

175 if self._released: 

176 raise ThrottleReleased( 

177 "throttle is released; cannot send updates", 

178 dcc_address=self.dcc_address, 

179 ) 

180 if self._throttle_id is None: 

181 raise RuntimeError( 

182 f"Throttle for DCC address {self.dcc_address} is not acquired; " 

183 "enter the 'async with' block before calling set_speed" 

184 ) 

185 if not (0.0 <= value <= 1.0): 

186 raise ValueError(f"speed value must be in [0.0, 1.0]; got {value!r}") 

187 await self._handle.throttle_update( 

188 self._throttle_id, 

189 {"speed": value, "forward": forward}, 

190 ) 

191 logger.info( 

192 "throttle speed updated", 

193 extra={ 

194 "dcc_address": self.dcc_address, 

195 "throttle_id": self._throttle_id, 

196 "speed": value, 

197 "forward": forward, 

198 }, 

199 )

201 async def set_function(self, n: int, on: bool) -> None: 

202 """Set one function bit via a fire-and-forget WS update (FR26). 

203 

204 Sends ``{"type":"throttle","data":{"throttle":<id>,"F<n>":<on>}}`` 

205 and returns as soon as the bytes are written; does NOT await JMRI's 

206 state-echo. 

207 

208 Higher function bits (F29+, used by some decoders such as 

209 ScaleTrains) are deferred to Growth (FR26). 

210 

211 Args: 

212 n: Function index in the closed range ``[0, 28]``. F0 is 

213 conventionally the headlight on most decoders. 

214 on: ``True`` to assert the bit, ``False`` to clear it. 

215 

216 Raises: 

217 ThrottleReleased: when called after :meth:`release` (carries 

218 ``dcc_address`` in ``.context``). No WS traffic is sent. 

219 RuntimeError: when called on a never-acquired ``Throttle``. 

220 ValueError: when ``n`` is outside ``[0, 28]``. 

221 JMRIConnectionError: surfaced from the WS transport. 

222 """ 

223 if self._released: 

224 raise ThrottleReleased( 

225 "throttle is released; cannot send updates", 

226 dcc_address=self.dcc_address, 

227 ) 

228 if self._throttle_id is None: 

229 raise RuntimeError( 

230 f"Throttle for DCC address {self.dcc_address} is not acquired; " 

231 "enter the 'async with' block before calling set_function" 

232 ) 

233 if isinstance(n, bool) or not isinstance(n, int): 

234 raise ValueError(f"function bit n must be an int; got {n!r}") 

235 if not (0 <= n <= 28): 

236 raise ValueError(f"function bit n must be in [0, 28]; got {n!r}") 

237 await self._handle.throttle_update( 

238 self._throttle_id, 

239 {f"F{n}": on}, 

240 ) 

241 logger.info( 

242 "throttle function updated", 

243 extra={ 

244 "dcc_address": self.dcc_address, 

245 "throttle_id": self._throttle_id, 

246 "function": n, 

247 "on": on, 

248 }, 

249 )

251 async def _release_impl(self, *, suppress_errors: bool) -> None: 

252 """Shared teardown: cancel keep-alive, send WS release, mark released. 

253 

254 Args: 

255 suppress_errors: When ``True`` (called from ``__aexit__`` while 

256 a body exception is propagating), log + swallow any WS 

257 release failure so the body exception keeps priority. When 

258 ``False`` (called from explicit ``release()``), let release 

259 errors propagate to the caller. 

260 """ 

261 # Cancel the keep-alive first so it can't fight us during release. 

262 if self._keepalive_task is not None and not self._keepalive_task.done(): 

263 self._keepalive_task.cancel() 

264 # Python's task machinery re-raises CancelledError when awaiting 

265 # a cancelled task even if the coroutine caught it internally 

266 # (asyncio sets Task._must_cancel; the StopIteration path then 

267 # calls super().cancel() before super().set_result()). 

268 try: 

269 await self._keepalive_task 

270 except asyncio.CancelledError: 

271 pass 

272 except Exception as e: 

273 logger.warning( 

274 "keepalive task raised during shutdown (swallowed)", 

275 extra={ 

276 "dcc_address": self.dcc_address, 

277 "throttle_id": self._throttle_id, 

278 "error_type": type(e).__name__, 

279 }, 

280 ) 

281 self._keepalive_task = None 

282 # Send the release. Fire-and-forget per AC4 — JMRI's release-echo is 

283 # consumed by the WS dispatcher but not awaited. 

284 if self._throttle_id is not None: 

285 try: 

286 await self._handle.throttle_release(self._throttle_id) 

287 except Exception as e: 

288 if suppress_errors: 

289 logger.warning( 

290 "throttle release failed during exception-driven teardown (swallowed)", 

291 extra={ 

292 "dcc_address": self.dcc_address, 

293 "throttle_id": self._throttle_id, 

294 "error_type": type(e).__name__, 

295 }, 

296 ) 

297 else: 

298 self._released = True 

299 logger.info( 

300 "throttle released (WS release errored)", 

301 extra={ 

302 "dcc_address": self.dcc_address, 

303 "throttle_id": self._throttle_id, 

304 }, 

305 ) 

306 raise 

307 self._released = True 

308 logger.info( 

309 "throttle released", 

310 extra={ 

311 "dcc_address": self.dcc_address, 

312 "throttle_id": self._throttle_id, 

313 }, 

314 )

316 # Story 5.3 hookpoint: the keep-alive body is a no-op stub. 

317 # 

318 # Confirmed unnecessary on JMRI 5.14 simulator / 2026-05-21 — JMRI's 

319 # WS-level heartbeat (server-advertised 13500 ms; pyjmri transport's 

320 # 10 s ping_interval) keeps the WS connection alive, which in turn 

321 # keeps all held throttles alive. The supervised task here exists 

322 # structurally per architecture sec. Concurrency Model (line 543-547): 

323 # "Per-throttle keep-alive coroutine, one per active Throttle". 

324 # 

325 # Confirmed unnecessary on JMRI 5.14.1 / NCE USB / 2026-05-26 by Mikey 

326 # — hardware-mode validation per CONTRIBUTING.md release checklist: 

327 # 30 s silent hold did not drop the throttle; subsequent set_speed(0.1) 

328 # moved DCC 1032 on real hardware. The simulator finding above extends 

329 # to hardware-mode on this JMRI x NCE USB combination. 

330 async def _keepalive(self) -> None: 

331 """No-op supervised stub; cancels cleanly on release.""" 

332 try: 

333 await asyncio.Event().wait() 

334 except asyncio.CancelledError: 

335 return
