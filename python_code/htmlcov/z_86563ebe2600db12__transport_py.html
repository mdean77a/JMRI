49class HTTPClient: 

50 """Async HTTP client wrapper around :class:`httpx.AsyncClient`. 

51 

52 Catches ``httpx.*`` exceptions and re-raises them as 

53 :class:`~pyjmri.exceptions.JMRIError` subclasses with ``from e``. No 

54 other module in the package is allowed to do this. 

55 

56 Args: 

57 host: JMRI host (e.g. ``"localhost"``). 

58 port: JMRI web-server port (e.g. ``12080``). 

59 scheme: Either ``"http"`` or ``"https"``. Defaults to ``"http"``. 

60 request_timeout: Per-request timeout in seconds. 

61 transport: Optional custom :class:`httpx.AsyncBaseTransport` for 

62 tests. Production callers leave this as ``None``. 

63 """ 

64 

65 def __init__( 

66 self, 

67 *, 

68 host: str, 

69 port: int, 

70 scheme: str = "http", 

71 request_timeout: float = 10.0, 

72 transport: httpx.AsyncBaseTransport | None = None, 

73 ) -> None: 

74 self._host = host 

75 self._port = port 

76 self._base_url = f"{scheme}://{host}:{port}" 

77 self._http = httpx.AsyncClient( 

78 base_url=self._base_url, 

79 timeout=request_timeout, 

80 transport=transport, 

81 )

83 async def get(self, path: str) -> dict[str, Any] | list[dict[str, Any]]: 

84 """GET ``path`` and return the parsed JSON body. 

85 

86 JMRI JSON v5 endpoints return either a single object 

87 (``GET /json/v5/<type>/<name>``) or an array of objects 

88 (``GET /json/v5/<type>``). Both shapes are returned as-is for 

89 the parsing layer (Story 2.2) to narrow. 

90 

91 Raises: 

92 JMRIConnectionError: when the underlying socket cannot connect, 

93 or any other transport-level failure occurs (network errors, 

94 proxy errors, protocol errors at the transport layer). 

95 JMRIRequestTimeout: when the request exceeds ``request_timeout``. 

96 JMRIProtocolError: when the response is non-200, not JSON, or 

97 does not match the JMRI shape (object or array of objects). 

98 """ 

99 try: 

100 response = await self._http.get(path) 

101 except httpx.ConnectError as e: 

102 raise JMRIConnectionError(host=self._host, port=self._port) from e 

103 except httpx.TimeoutException as e: 

104 raise JMRIRequestTimeout( 

105 "HTTP request exceeded request_timeout", 

106 host=self._host, 

107 port=self._port, 

108 path=path, 

109 ) from e 

110 except httpx.TransportError as e: 

111 # Covers ReadError, WriteError, CloseError, ProxyError, 

112 # ProtocolError, DecodingError, TooManyRedirects, etc. 

113 raise JMRIConnectionError( 

114 host=self._host, 

115 port=self._port, 

116 error_type=type(e).__name__, 

117 ) from e 

118 

119 if response.status_code != 200: 

120 raise JMRIProtocolError( 

121 "unexpected HTTP status", 

122 status=response.status_code, 

123 path=path, 

124 ) 

125 

126 try: 

127 payload = response.json() 

128 except ValueError as e: 

129 raise JMRIProtocolError( 

130 "response body was not valid JSON", 

131 path=path, 

132 ) from e 

133 

134 if isinstance(payload, dict) or ( 

135 isinstance(payload, list) and all(isinstance(item, dict) for item in payload) 

136 ): 

137 logger.debug( 

138 "HTTP GET %s -> %d", 

139 path, 

140 response.status_code, 

141 extra={"method": "GET", "path": path, "status": response.status_code}, 

142 ) 

143 return payload 

144 

145 raise JMRIProtocolError( 

146 "response JSON was not a JMRI v5 entity (object or array of objects)", 

147 path=path, 

148 actual_type=type(payload).__name__, 

149 )

151 async def command( 

152 self, 

153 entity_type: str, 

154 name: str, 

155 payload: dict[str, Any], 

156 ) -> None: 

157 """POST a state/value command to JMRI and return when JMRI acks. 

158 

159 FR22 / FR37 discipline: JMRI's HTTP 2xx response means JMRI 

160 accepted the command into its model. It does NOT mean the 

161 physical layout has reached the commanded state — NCE is 

162 open-loop and the library never claims a state it did not 

163 observe. Story 4.2 layers a ``wait_for_jmri_state=True`` opt-in 

164 over this method for callers who want to await the WS echo. 

165 

166 Args: 

167 entity_type: JMRI entity-type string (``"turnout"``, 

168 ``"light"``, ``"memory"``, ``"route"``). 

169 name: Entity system name; URL-encoded into the path. 

170 payload: Per-type body fragment merged into 

171 ``{"type", "data": {"name", **payload}}``. For state- 

172 bearing types pass ``{"state": <int>}``; for memory pass 

173 ``{"value": <str>}``. 

174 

175 Raises: 

176 JMRIConnectionError: socket-level failure (transport error). 

177 JMRIRequestTimeout: request exceeded ``request_timeout``. 

178 JMRIProtocolError: JMRI returned a non-2xx with a non-error- 

179 envelope body, or a 2xx with a malformed body. 

180 LayoutEntityNotControllable: JMRI returned an error envelope 

181 indicating the entity is locked, refuses the command, or 

182 is otherwise not controllable. 

183 LayoutEntityNotFound: JMRI returned an error envelope 

184 indicating the entity name does not exist on JMRI's 

185 side. 

186 """ 

187 path = f"/json/v5/{entity_type}/{quote(name, safe='')}" 

188 body = {"type": entity_type, "data": {"name": name, **payload}} 

189 try: 

190 response = await self._http.post(path, json=body) 

191 except httpx.ConnectError as e: 

192 raise JMRIConnectionError(host=self._host, port=self._port) from e 

193 except httpx.TimeoutException as e: 

194 raise JMRIRequestTimeout( 

195 "HTTP request exceeded request_timeout", 

196 host=self._host, 

197 port=self._port, 

198 path=path, 

199 ) from e 

200 except httpx.TransportError as e: 

201 raise JMRIConnectionError( 

202 host=self._host, 

203 port=self._port, 

204 error_type=type(e).__name__, 

205 ) from e 

206 

207 status = response.status_code 

208 if 200 <= status < 300: 

209 logger.debug( 

210 "HTTP POST %s -> %d", 

211 path, 

212 status, 

213 extra={"method": "POST", "path": path, "status": status}, 

214 ) 

215 return None 

216 

217 # Non-2xx — try to translate JMRI's error envelope. 

218 jmri_message = _extract_jmri_error_message(response) 

219 if jmri_message is None: 

220 raise JMRIProtocolError( 

221 "unexpected HTTP status from command", 

222 status=status, 

223 path=path, 

224 ) 

225 if status == 404: 

226 raise LayoutEntityNotFound( 

227 entity_type=entity_type, 

228 name=name, 

229 jmri_message=jmri_message, 

230 status=status, 

231 ) 

232 if status in (400, 403, 409): 

233 raise LayoutEntityNotControllable( 

234 entity_type=entity_type, 

235 name=name, 

236 jmri_message=jmri_message, 

237 status=status, 

238 ) 

239 raise JMRIProtocolError( 

240 "unexpected HTTP status from command", 

241 status=status, 

242 path=path, 

243 jmri_message=jmri_message, 

244 )

251class WSConnection: 

252 """WebSocket connection wrapper consuming ``websockets.connect()`` as an async iterator. 

253 

254 Architecture sec. Transport Layer + sec. Reconnect & Restoration 

255 Mechanism. Only this module imports ``websockets``. 

256 

257 The library's built-in ``async for ... in connect(...)`` iterator drives 

258 reconnection. pyjmri does not write its own retry loop. The 

259 ``process_exception`` hook returns the original exception once 

260 ``ReconnectConfig.max_attempts`` consecutive failures occur, causing 

261 the iterator to raise; :meth:`run` then re-raises as 

262 :class:`~pyjmri.exceptions.JMRIReconnectFailed`. With 

263 ``max_attempts=None`` (the default), the hook returns ``None`` 

264 forever and retries continue indefinitely. 

265 

266 Note: in ``websockets`` v16+, returning ``None`` from ``process_exception`` 

267 means retryable (inverted from earlier library versions where ``None`` 

268 meant give up). Full hook contract: :meth:`_process_exception`. 

269 

270 Args: 

271 host: JMRI host (e.g. ``"localhost"``). 

272 port: JMRI web-server port (e.g. ``12080``). 

273 reconnect_config: Reconnect tuning. Currently only 

274 ``max_attempts`` is honored — delay timing is delegated to 

275 the ``websockets`` library's built-in ``backoff()``. 

276 secure: When ``True``, use ``wss://`` (TLS). :class:`~pyjmri.client.Client` 

277 sets this from its HTTP scheme (``https`` / ``wss`` URLs → secure). 

278 on_reconnect: Optional coroutine invoked after each successful 

279 reconnect (not the very first connect). The Client passes 

280 ``SubscriptionRegistry.replay`` here. 

281 connected_event: Optional :class:`asyncio.Event` set once on the 

282 very first successful connect. The Client awaits this in 

283 ``__aenter__`` so users see a fully-open Client. 

284 """ 

285 

286 def __init__( 

287 self, 

288 *, 

289 host: str, 

290 port: int, 

291 reconnect_config: ReconnectConfig, 

292 secure: bool = False, 

293 on_reconnect: Callable[[], Awaitable[None]] | None = None, 

294 connected_event: Any = None, # asyncio.Event, typed Any to avoid eager import 

295 ) -> None: 

296 self._host = host 

297 self._port = port 

298 self._secure = secure 

299 self._config = reconnect_config 

300 self._on_reconnect = on_reconnect 

301 self._connected = connected_event 

302 self._connection: ClientConnection | None = None 

303 self._attempt: int = 0 

304 self._give_up_cause: BaseException | None = None

309 return f"{proto}://{self._host}:{self._port}{WS_PATH}"

311 async def run(self, on_message: Callable[[dict[str, Any]], Awaitable[None]]) -> None: 

312 """Long-running coroutine: connect, receive, reconnect indefinitely. 

313 

314 Supervised by the Client's :class:`asyncio.TaskGroup`. Cancellation 

315 propagates from the TaskGroup's ``__aexit__`` and exits this 

316 coroutine cleanly. 

317 """ 

318 try: 

319 async for connection in websockets.connect( 

320 self.url, 

321 process_exception=self._process_exception, 

322 ping_interval=_PING_INTERVAL_SEC, 

323 ): 

324 self._connection = connection 

325 first_connect = self._connected is not None and not self._connected.is_set() 

326 self._attempt = 0 # reset on each successful connect 

327 if first_connect: 

328 self._connected.set() 

329 else: 

330 if self._on_reconnect is not None: 

331 try: 

332 await self._on_reconnect() 

333 except Exception as cb_exc: 

334 logger_reconnect.warning( 

335 "on_reconnect callback raised; continuing receive loop", 

336 extra={ 

337 "host": self._host, 

338 "error_type": type(cb_exc).__name__, 

339 }, 

340 ) 

341 try: 

342 async for raw in connection: 

343 try: 

344 envelope = _decode_ws_frame(raw) 

345 logger.debug( 

346 "WS inbound", 

347 extra={ 

348 "direction": "inbound", 

349 "type": envelope.get("type"), 

350 }, 

351 ) 

352 await on_message(envelope) 

353 except Exception as msg_exc: 

354 logger.warning( 

355 "WS message handling error; skipping frame", 

356 extra={"error_type": type(msg_exc).__name__}, 

357 ) 

358 except WebSocketException: 

359 self._connection = None 

360 continue 

361 except WebSocketException as e: 

362 self._connection = None 

363 if self._give_up_cause is not None: 

364 raise JMRIReconnectFailed( 

365 host=self._host, 

366 port=self._port, 

367 attempts=self._attempt, 

368 cause=type(self._give_up_cause).__name__, 

369 ) from self._give_up_cause 

370 raise JMRIConnectionError(host=self._host, port=self._port) from e

372 def _process_exception(self, exc: Exception) -> Exception | None: 

373 """``websockets`` v16 retry hook. 

374 

375 Returns ``None`` to mark the exception retryable (library schedules 

376 the next reconnect via its built-in ``backoff()``). Returns the 

377 exception itself to mark it fatal (library raises it and exits the 

378 async iterator). 

379 """ 

380 self._attempt += 1 

381 if self._config.max_attempts is not None and self._attempt >= self._config.max_attempts: 

382 self._give_up_cause = exc 

383 logger_reconnect.warning( 

384 "WebSocket reconnect attempts exhausted", 

385 extra={ 

386 "host": self._host, 

387 "attempt": self._attempt, 

388 "max_attempts": self._config.max_attempts, 

389 "error_type": type(exc).__name__, 

390 }, 

391 ) 

392 return exc 

393 logger_reconnect.warning( 

394 "WebSocket disconnected; will retry", 

395 extra={ 

396 "host": self._host, 

397 "attempt": self._attempt, 

398 "error_type": type(exc).__name__, 

399 }, 

400 ) 

401 return None

403 async def _force_disconnect(self) -> None: 

404 """Test-only hook: close the active WS connection to trigger the reconnect loop. 

405 

406 Closing ``self._connection`` causes the ``async for raw in connection`` 

407 inner loop to exit, which the outer ``async for connection in 

408 websockets.connect(...)`` iterator handles by scheduling a reconnect. 

409 Safe to call when no connection is active (no-op). Never call from 

410 production code — use this hook only in integration tests. 

411 """ 

412 if self._connection is not None: 

413 await self._connection.close()

415 async def send(self, message: dict[str, Any]) -> None: 

416 """JSON-encode ``message`` and send it on the current connection. 

417 

418 Raises :class:`JMRIConnectionError` if no connection is active 

419 (e.g., a send arrives between disconnect and reconnect, or after 

420 the supervisor task has been cancelled). 

421 """ 

422 connection = self._connection 

423 if connection is None: 

424 raise JMRIConnectionError(host=self._host, port=self._port) 

425 try: 

426 await connection.send(json.dumps(message)) 

427 except WebSocketException as e: 

428 raise JMRIConnectionError(host=self._host, port=self._port) from e 

429 logger.debug( 

430 "WS outbound", 

431 extra={"direction": "outbound", "type": message.get("type")}, 

432 )

435def _extract_jmri_error_message(response: httpx.Response) -> str | None: 

436 """Return JMRI's error message if the body matches the error envelope. 

437 

438 JMRI 5.14+ returns ``{"type":"error","data":{"code":<int>, 

439 "message":"<text>"}}`` for command-path failures. Returns ``None`` 

440 when the body cannot be parsed as JSON or does not match this shape; 

441 the caller then raises :class:`JMRIProtocolError` instead. 

442 """ 

443 try: 

444 body = response.json() 

445 except (ValueError, UnicodeDecodeError): 

446 return None 

447 if not isinstance(body, dict): 

448 return None 

449 if body.get("type") != "error": 

450 return None 

451 data = body.get("data") 

452 if not isinstance(data, dict): 

453 return None 

454 message = data.get("message") 

455 if not isinstance(message, str): 

456 return None 

457 return message

460def _decode_ws_frame(raw: Any) -> dict[str, Any]: 

461 """Decode a WebSocket text or binary frame into a JMRI envelope dict. 

462 

463 Raises: 

464 JMRIProtocolError: when the frame is not valid JSON or is not a 

465 JSON object. 

466 """ 

467 if isinstance(raw, (bytes, bytearray)): 

468 raw_text = raw.decode("utf-8") 

469 elif isinstance(raw, str): 

470 raw_text = raw 

471 else: 

472 raise JMRIProtocolError( 

473 "unexpected WebSocket frame type", 

474 frame_type=type(raw).__name__, 

475 ) 

476 try: 

477 envelope = json.loads(raw_text) 

478 except ValueError as e: 

479 raise JMRIProtocolError("WebSocket frame was not valid JSON") from e 

480 if not isinstance(envelope, dict): 

481 raise JMRIProtocolError( 

482 "WebSocket envelope was not a JSON object", 

483 actual_type=type(envelope).__name__, 

484 ) 

485 return envelope
